# 云服务器迁移配置指南

> 适用场景：更换阿里云 ECS 实例、重装系统、迁移到其他云服务商

---

## 一、服务器基础环境

### 1.1 操作系统

推荐 Alibaba Cloud Linux 3 / CentOS 8+ / Ubuntu 20.04+

### 1.2 安装 Docker

```bash
# 安装 Docker
curl -fsSL https://get.docker.com | bash
systemctl enable docker
systemctl start docker

# 配置镜像加速（阿里云）
mkdir -p /etc/docker
cat > /etc/docker/daemon.json << 'EOF'
{
  "registry-mirrors": ["https://50w6cpne.mirror.aliyuncs.com"]
}
EOF
systemctl restart docker
```

### 1.3 目录规划

```bash
mkdir -p /opt/smart-lighting/{app,logs}
mkdir -p /opt/smart-lighting/mqtt/emqx/{data,log,etc,certs}
mkdir -p /data/tdengine
```

## 二、安全组（防火墙）配置

在云控制台安全组入方向添加：

| 端口 | 协议 | 用途 |
|------|------|------|
| `22` | TCP | SSH 远程管理 |
| `1883` | TCP | MQTT TCP 连接 |
| `8883` | TCP | MQTT SSL 连接 |
| `8083` | TCP | WebSocket 连接 |
| `8084` | TCP | WebSocket SSL 连接 |
| `18083` | TCP | EMQX Dashboard |
| `6030` | TCP | TDengine Native 连接 |
| `6041` | TCP | TDengine REST 接口 |
| `6060` | TCP | TDengine Web 探勘器 |
| `8080` | TCP | 应用端口 / MaxKB |

## 四、TDengine 时序数据库

### 4.1 启动容器

```bash
docker run -d --name tdengine \
  --restart always \
  --network iot-net \
  --hostname tdengine \
  -p 6030:6030 -p 6041:6041 -p 6060:6060 \
  -v /data/tdengine:/var/lib/taos \
  -e TAOS_FQDN=tdengine \
  tdengine/tdengine:3.3.2.0
```

### 4.2 修复 Web 探勘器

```bash
docker exec tdengine sed -i 's/buildkitsandbox/127.0.0.1/g' /etc/taos/explorer.toml
docker restart tdengine
```

### 4.3 创建数据库和超级表

```bash
# 进入 CLI
docker exec -it tdengine taos

# 在 taos CLI 中执行：
CREATE DATABASE smart_lighting KEEP 365;
USE smart_lighting;

CREATE STABLE telemetry (
    ts TIMESTAMP, illuminance DOUBLE, temperature DOUBLE,
    humidity DOUBLE, pm25 DOUBLE, aqi INT, pir TINYINT, traffic_flow INT
) TAGS (device_id VARCHAR(50));

CREATE STABLE vision_event (
    ts TIMESTAMP, event_type VARCHAR(30),
    confidence DOUBLE, snapshot_ref VARCHAR(255)
) TAGS (device_id VARCHAR(50));

CREATE STABLE voice_event (
    ts TIMESTAMP, type VARCHAR(20),
    content VARCHAR(500), source VARCHAR(20)
) TAGS (device_id VARCHAR(50));
```

## 五、EMQX 消息服务器

### 5.1 启动容器

```bash
docker run -d --name emqx \
  --restart always \
  --network iot-net \
  -p 1883:1883 -p 8883:8883 \
  -p 8083:8083 -p 8084:8084 \
  -p 18083:18083 \
  emqx/emqx:6.2.1
```

### 5.2 初始配置（通过 REST API）

以下配置通过 EMQX REST API 完成，可在 Postman 或 curl 中批量执行：

```bash
# === 变量 ===
EMQX="http://47.96.27.141:18083/api/v5"
ADMIN_PWD="xyx246824XYX@"  # 替换为你的管理员密码

# === 1. 登录获取 Token ===
TOKEN=$(curl -s -u "admin:public" -X POST "$EMQX/login" \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"public"}' | sed 's/.*"token":"\([^"]*\)".*/\1/')

# === 2. 修改管理员密码 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X PUT "$EMQX/users/admin" \
  -d "{\"password\":\"$ADMIN_PWD\",\"description\":\"管理员\"}"

# 重新登录
TOKEN=$(curl -s -u "admin:$ADMIN_PWD" -X POST "$EMQX/login" \
  -H "Content-Type: application/json" \
  -d "{\"username\":\"admin\",\"password\":\"$ADMIN_PWD\"}" | sed 's/.*"token":"\([^"]*\)".*/\1/')

# === 3. 创建内置数据库认证器 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/authentication" \
  -d '{
    "mechanism": "password_based",
    "backend": "built_in_database",
    "enable": true,
    "password_hash_algorithm": {"name": "plain", "salt_position": "disable"},
    "user_id_type": "username"
  }'

# === 4. 添加服务账号 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/authentication/password_based:built_in_database/users" \
  -d '{"user_id":"backend","password":"123456"}'

# === 5. 更新 ACL 规则 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X PUT "$EMQX/authorization/sources/file" \
  -d '{
    "enable": true,
    "type": "file",
    "rules": "{allow, {username, \"backend\"}, all, [\"streetlight/#\", \"system/#\"]}.\n{allow, {username, \"dashboard\"}, subscribe, [\"$SYS/#\"]}.\n{allow, all, publish, [\"streetlight/${username}/telemetry\", \"streetlight/${username}/heartbeat\", \"streetlight/${username}/vision/event\", \"streetlight/${username}/voice/event\", \"streetlight/${username}/command/ack\"]}.\n{allow, all, subscribe, [\"streetlight/${username}/control\"]}.\n{deny, all}."
  }'

# === 6. 创建 TDengine 连接器 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/connectors" \
  -d '{
    "type": "tdengine",
    "name": "tdengine",
    "server": "tdengine:6041",
    "username": "root",
    "password": "taosdata",
    "pool_size": 4
  }'

# === 7. 创建遥测桥接规则 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/rules" \
  -d '{
    "id": "telemetry_bridge",
    "sql": "SELECT payload.illuminance as illuminance, payload.temperature as temperature, payload.humidity as humidity, payload.pm25 as pm25, payload.aqi as aqi, CASE WHEN payload.pir >= -128 AND payload.pir <= 127 THEN payload.pir ELSE NULL END as pir, payload.trafficFlow as traffic_flow, payload.collectedAt as ts, nth(2, split(topic, '"'"'/'"'"')) as dev_id FROM \"streetlight/+/telemetry\"",
    "enable": true,
    "actions": ["tdengine:telemetry_action"]
  }'

curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/actions" \
  -d '{
    "type": "tdengine",
    "connector": "tdengine",
    "name": "telemetry_action",
    "parameters": {
      "database": "smart_lighting",
      "sql": "INSERT INTO t_${dev_id} USING telemetry TAGS ('"'"'${dev_id}'"'"') VALUES ('"'"'${ts}'"'"', ${illuminance}, ${temperature}, ${humidity}, ${pm25}, ${aqi}, ${pir}, ${traffic_flow})",
      "undefined_vars_as_null": true
    },
    "enable": true
  }'

# === 8. 创建视觉事件桥接规则 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/rules" \
  -d '{
    "id": "vision_bridge",
    "sql": "SELECT payload.eventType as event_type, payload.confidence as confidence, payload.snapshotRef as snapshot_ref, payload.occurredAt as ts, nth(2, split(topic, '"'"'/'"'"')) as dev_id FROM \"streetlight/+/vision/event\"",
    "enable": true,
    "actions": ["tdengine:vision_action"]
  }'

curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/actions" \
  -d '{
    "type": "tdengine",
    "connector": "tdengine",
    "name": "vision_action",
    "parameters": {
      "database": "smart_lighting",
      "sql": "INSERT INTO vis_${dev_id} USING vision_event TAGS ('"'"'${dev_id}'"'"') VALUES ('"'"'${ts}'"'"', '"'"'${event_type}'"'"', ${confidence}, '"'"'${snapshot_ref}'"'"')",
      "undefined_vars_as_null": true
    },
    "enable": true
  }'

# === 9. 创建语音事件桥接规则 ===
curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/rules" \
  -d '{
    "id": "voice_bridge",
    "sql": "SELECT payload.type as type, payload.content as content, payload.source as source, payload.occurredAt as ts, nth(2, split(topic, '"'"'/'"'"')) as dev_id FROM \"streetlight/+/voice/event\"",
    "enable": true,
    "actions": ["tdengine:voice_action"]
  }'

curl -s -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -X POST "$EMQX/actions" \
  -d '{
    "type": "tdengine",
    "connector": "tdengine",
    "name": "voice_action",
    "parameters": {
      "database": "smart_lighting",
      "sql": "INSERT INTO voi_${dev_id} USING voice_event TAGS ('"'"'${dev_id}'"'"') VALUES ('"'"'${ts}'"'"', '"'"'${type}'"'"', '"'"'${content}'"'"', '"'"'${source}'"'"')",
      "undefined_vars_as_null": true
    },
    "enable": true
  }'
```

> **注意**：规则中使用 `nth(2, split(topic, '/'))` 提取 deviceId，**不能**使用 `payload.deviceId`（EMQX 6.2.1 bug）。

## 九、系统调优

### 文件描述符限制

```bash

echo "fs.file-max = 524288" >> /etc/sysctl.conf

```

| 检查项 | 命令 | 预期 |
|--------|------|------|
